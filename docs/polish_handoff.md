```python
"""Extract the polish-layer source embedded in the register workbook."""
from openpyxl import load_workbook
from pathlib import Path
import sys

SHEET = "Reference (polish)"
TARGETS = [
    ("FILE 1 of 3",     "config/polish_spec.yaml"),
    ("FILE 2 of 3",     "tools/register_polish.py"),
    ("FILE 3 of 3",     "tools/polish_charts.py"),
    ("FILE 2 ADDENDUM", "tools/_polish_inbox_addendum.py"),
]


def extract(xlsx):
    ws = load_workbook(xlsx).get_sheet_by_name(SHEET)
    lines = [(c.row, str(c.value) if c.value is not None else "")
             for c in ws["A"]]
    starts = {}
    for row, text in lines:
        for marker, path in TARGETS:
            if text.startswith(marker):
                starts[row] = path
    bounds = sorted(starts)
    for i, start in enumerate(bounds):
        end = bounds[i + 1] - 2 if i + 1 < len(bounds) else len(lines)
        body = [t for r, t in lines if start + 2 <= r < end]
        while body and not body[0].strip():
            body.pop(0)
        while body and not body[-1].strip():
            body.pop()
        out = Path(starts[start])
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_text("\n".join(body) + "\n")
        print(f"wrote {out}  ({len(body)} lines)")


if __name__ == "__main__":
    extract(sys.argv[1] if len(sys.argv) > 1 else "dist/wind_register.xlsx")
```

three iterations. None of the shipped code has ever been executed.

## Stage 2 - idempotency
Run twice on the same file; the second run must change nothing. Do NOT hash the .xlsx - zip
timestamps differ on every write. Compare a normalised dump.

## Stage 3 - fidelity  (the one that matters)
Same dump, module output vs tests/golden/. Stages 1-2 prove the code runs; only Stage 3 proves
it reproduces the workbook that was actually approved.
Expect genuine diffs on column widths: autofit() approximates from character count and will not
match Excel's renderer. Either allow a tolerance on width, or freeze the offenders in widths:.

```python
"""tests/test_polish.py"""
import pytest
from openpyxl import load_workbook
from tools.register_polish import polish

GOLDEN = "tests/golden/wind_register_polished.xlsx"


def dump(path):
    """Normalised comparable view - values, formats, fills, charts, widths."""
    wb = load_workbook(path)
    out = []
    for ws in wb.worksheets:
        for row in ws.iter_rows():
            for c in row:
                if c.value is not None or c.number_format != "General":
                    out.append((ws.title, c.coordinate, str(c.value),
                                c.number_format,
                                getattr(c.fill.fgColor, "rgb", None),
                                c.font.b))
        out.append((ws.title, "_charts", sorted(str(ch.title) for ch in ws._charts)))
    return out


def widths(path):
    wb = load_workbook(path)
    return {(ws.title, k): round(v.width or 0)
            for ws in wb.worksheets for k, v in ws.column_dimensions.items()}


@pytest.fixture
def polished(tmp_path, scaffolded):        # scaffolded = your existing build fixture
    target = tmp_path / "reg.xlsx"
    target.write_bytes(scaffolded.read_bytes())
    polish(str(target))
    return str(target)


def test_smoke(polished):
    assert load_workbook(polished)["Charts"]._charts


def test_idempotent(polished):
    before = dump(polished)
    polish(polished)
    assert dump(polished) == before


def test_fidelity(polished):
    assert dump(polished) == dump(GOLDEN)


@pytest.mark.xfail(reason="autofit is a character-count approximation, not Excel's renderer")
def test_widths_match_golden(polished):
    assert widths(polished) == widths(GOLDEN)


def test_invariants(polished):
    wb = load_workbook(polished)

    # 4 - P-level categories must be TEXT or they chart as a data series
    assert isinstance(wb["Charts"]["AA3"].value, str)
    assert isinstance(wb["Charts"]["AD3"].value, str)

    # 3 - Lists column headers are in ROW 2; row 1 is the cascade note
    assert (wb["Lists"]["A2"].fill.fgColor.rgb or "").endswith("1F4E79")
    assert not (wb["Lists"]["A1"].fill.fgColor.rgb or "").endswith("1F4E79")

    # 6 - tornado: largest risk at the TOP
    t = next(c for c in wb["Charts"]._charts if "tornado" in str(c.title).lower())
    assert t.y_axis.scaling.orientation == "maxMin"

    # 7 - Charts unprotected (protection kills the column-outline toggle)
    assert not wb["Charts"].protection.sheet
    assert wb["Build Log"].protection.sheet

    # 2 - no font colour written on data cells by the polish pass
    rgb = getattr(wb["Risk Register"]["P5"].font.color, "rgb", None)
    assert rgb in (None, "FF000000")

    # 5 charts present
    assert len(wb["Charts"]._charts) == 5


def test_inbox_merge(polished, tmp_path):
    """Flip the shipped example row to pending; it must merge exactly once."""
    wb = load_workbook(polished)
    wb["Polish Inbox"]["G12"] = "pending"
    wb.save(polished)

    polish(polished)
    wb = load_workbook(polished)
    assert wb["Polish Inbox"]["G12"].value == "merged"
    assert str(wb["Polish Inbox"]["H12"].value).startswith("polish-v")
    assert len(wb["Charts"]._charts) == 6

    polish(polished)                       # merge must not duplicate
    assert len(load_workbook(polished)["Charts"]._charts) == 6
```

## Two gotchas that WILL cost an hour if you do not know them

1. openpyxl DOES NOT CALCULATE FORMULAS. It writes the formula string with no cached result.
   A freshly polished file from CI has None behind every formula cell until Excel opens it and
   recalculates. Concretely: the Charts helper columns are all formulas, so THE CHARTS WILL
   LOOK EMPTY in anything that does not recalc - LibreOffice preview, headless PDF export,
   pandas.read_excel, load_workbook(data_only=True). Open once in Excel and they populate.
   This is expected behaviour, not a bug. Do not "fix" it by hardcoding values.

2. The golden-file test MUST use data_only=False (the default). With data_only=True both sides
   read None, every comparison trivially matches, and the suite goes green while proving
   nothing. Worst possible failure mode.

If calculated values are genuinely needed headlessly: LibreOffice --headless --convert-to xlsx
forces a recalc, or use the `formulas` library. Both are heavier than they sound - avoid unless
there is a real requirement.

# RULES - do not violate these
===============================================================================================

1.  openpyxl has NO autofit. Widths are explicit, or via the autofit() emulator with
    ignore_rows for note rows that sit inside a data column.
2.  NEVER write font colour. blue = input, black = formula, amber fill = model write-back,
    thin blue border = model-read. The TOC cell-type legend depends on it. Polisher._guard()
    enforces this - do not weaken it.
3.  Lists column headers are in ROW 2. Row 1 is the cascade note.
4.  Numeric P-level columns must be TEXT to chart as categories, else Excel makes them a series.
5.  Scatter needs X (cost) LEFT of Y (probability). The Simulation source has them reversed;
    that is what the Charts!X:Y helper block is for.
6.  Tornado needs a reversed category axis (maxMin) so the largest risk is at the top.
7.  Charts tab stays UNPROTECTED. Sheet protection disables the column-outline +/- toggle
    (EnableOutlining is VBA-only).
8.  Simulation!H4 is PERCENT-AS-NUMBER (0.1189 = 0.119%), not a fraction. VERIFY THIS against
    the engine - it is the one value the formatting pass had to infer.
9.  Polish Inbox is NOT delete-then-recreate. It carries state (status / merged_in). Charts and
    Build Log are recreated every run; the inbox is not.
10. Sheet protection: unprotect -> format -> re-protect, workbook structure last.
